# RPG Tracker — полный код всех новых и изменённых файлов

Это копируемый листинг **именно готовой версии**. В каждом разделе указан путь относительно корня `rpg-tracker`. Создайте или замените файл целиком. Для запуска придерживайтесь последовательности из `README_RU.md`.

**Важно:** `.env.local` не показан, потому что в нём должны храниться ваши реальные настройки. Используйте шаблон `.env.local.example`. Не копируйте значения из чужого проекта.

### `supabase/01_setup.sql`

```sql
-- RPG Tracker — первоначальная схема и ограничения доступа.
-- ВАЖНО: сначала сохранить резервную копию и проверить существующую таблицу characters.
-- Скрипт пересоздаёт RLS-политики на таблицах приложения, но не удаляет персонажей.

create table if not exists public.characters (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  name text not null,
  max_hp integer not null default 20,
  current_hp integer not null default 20,
  armor integer not null default 0,
  image_url text,
  image_path text,
  status text not null default 'pending',
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
-- Добавляем отсутствующие поля, если таблица уже существовала.
alter table public.characters add column if not exists image_path text;
alter table public.characters add column if not exists updated_at timestamptz not null default now();
create index if not exists characters_user_idx on public.characters(user_id);
create index if not exists characters_status_idx on public.characters(status);

create table if not exists public.admins (
  user_id uuid primary key references auth.users(id) on delete cascade
);

create or replace function public.is_admin()
returns boolean
language sql stable security definer
set search_path = ''
as $$
  select exists (
    select 1 from public.admins
    where user_id = (select auth.uid())
  );
$$;
revoke all on function public.is_admin() from public;
grant execute on function public.is_admin() to anon, authenticated;

create table if not exists public.character_cards (
  id uuid primary key default gen_random_uuid(),
  character_id uuid not null references public.characters(id) on delete cascade,
  type text not null check (type in ('attack', 'defense', 'healing')),
  name text not null check (char_length(trim(name)) between 1 and 100),
  description text not null default '' check (char_length(description) <= 2000),
  created_at timestamptz not null default now()
);
create index if not exists cards_character_idx on public.character_cards(character_id);

create table if not exists public.character_actions (
  id uuid primary key default gen_random_uuid(),
  character_id uuid not null references public.characters(id) on delete cascade,
  action_type text not null,
  amount integer not null default 0,
  blocked_amount integer not null default 0,
  hp_before integer not null,
  hp_after integer not null,
  armor_before integer not null,
  armor_after integer not null,
  note text not null default '',
  performed_by uuid references auth.users(id) on delete set null,
  created_at timestamptz not null default now()
);
create index if not exists actions_character_created_idx
  on public.character_actions(character_id, created_at desc);

-- Приводим существующих персонажей к корректному формату, не меняя их владельцев.
update public.characters set updated_at = now() where updated_at is null;
-- Не исправляем молча некорректные max_hp/armor: сначала проверьте существующие данные.

alter table public.characters enable row level security;
alter table public.admins enable row level security;
alter table public.character_cards enable row level security;
alter table public.character_actions enable row level security;

-- Удаляем старые политики приложения, чтобы старое разрешение UPDATE не
-- позволяло игрокам самостоятельно менять HP или одобрять свои анкеты.
do $$
declare p record;
begin
  for p in
    select schemaname, tablename, policyname from pg_policies
    where schemaname = 'public'
      and tablename in ('characters', 'admins', 'character_cards', 'character_actions')
  loop
    execute format('drop policy %I on %I.%I', p.policyname, p.schemaname, p.tablename);
  end loop;
end $$;

-- У анонимных пользователей доступ только к одобренным персонажам.
create policy characters_read on public.characters
  for select to anon, authenticated
  using (status = 'approved' or user_id = (select auth.uid()) or public.is_admin());

-- Каждый игрок может создать только собственную заявку на модерации.
create policy characters_apply on public.characters
  for insert to authenticated
  with check (
    user_id = (select auth.uid())
    and status = 'pending'
    and char_length(trim(name)) between 1 and 100
    and max_hp between 1 and 100000
    and current_hp = max_hp
    and armor = 0
  );

create policy characters_admin_update on public.characters
  for update to authenticated
  using (public.is_admin()) with check (public.is_admin());

create policy characters_admin_delete on public.characters
  for delete to authenticated using (public.is_admin());

create policy admins_read_self on public.admins
  for select to authenticated using (user_id = (select auth.uid()));

create policy cards_read on public.character_cards
  for select to anon, authenticated
  using (exists (
    select 1 from public.characters c
    where c.id = character_id
      and (c.status = 'approved' or c.user_id = (select auth.uid()) or public.is_admin())
  ));
create policy cards_admin_insert on public.character_cards
  for insert to authenticated with check (public.is_admin());
create policy cards_admin_update on public.character_cards
  for update to authenticated using (public.is_admin()) with check (public.is_admin());
create policy cards_admin_delete on public.character_cards
  for delete to authenticated using (public.is_admin());

create policy actions_read on public.character_actions
  for select to anon, authenticated
  using (exists (
    select 1 from public.characters c
    where c.id = character_id
      and (c.status = 'approved' or c.user_id = (select auth.uid()) or public.is_admin())
  ));
-- Нет политик вставки/редактирования истории: записи создаёт только серверная функция.

-- Привилегии SQL поверх RLS: без доступа к таблице политика сама по себе не сработает.
grant select on public.characters, public.character_cards, public.character_actions to anon, authenticated;
grant insert, update, delete on public.characters, public.character_cards to authenticated;
revoke all on public.admins from anon, authenticated;
grant select on public.admins to authenticated;
revoke all on public.character_actions from anon, authenticated;
grant select on public.character_actions to anon, authenticated;

-- Атомарное применение урона, лечения и брони.
create or replace function public.apply_character_action(
  p_character_id uuid,
  p_action text,
  p_amount integer,
  p_note text default ''
)
returns void
language plpgsql security definer
set search_path = ''
as $$
declare
  c public.characters%rowtype;
  v_hp integer;
  v_armor integer;
  v_blocked integer := 0;
begin
  if not public.is_admin() then
    raise exception 'Недостаточно прав';
  end if;
  if p_action not in ('damage', 'heal', 'set_armor')
     or p_amount is null or p_amount < 0 or p_amount > 100000
     or (p_action in ('damage', 'heal') and p_amount = 0) then
    raise exception 'Некорректное действие или количество';
  end if;
  if char_length(coalesce(p_note, '')) > 500 then
    raise exception 'Комментарий слишком длинный';
  end if;

  select * into c from public.characters
    where id = p_character_id for update;
  if not found then raise exception 'Персонаж не найден'; end if;
  if c.status <> 'approved' then
    raise exception 'Изменять характеристики можно только у одобренного персонажа';
  end if;

  v_hp := c.current_hp;
  v_armor := c.armor;
  if p_action = 'damage' then
    v_blocked := least(c.armor, p_amount);
    v_armor := c.armor - v_blocked;
    v_hp := greatest(0, c.current_hp - (p_amount - v_blocked));
  elsif p_action = 'heal' then
    v_hp := least(c.max_hp, c.current_hp + p_amount);
  elsif p_action = 'set_armor' then
    v_armor := p_amount;
  end if;

  update public.characters
    set current_hp = v_hp, armor = v_armor, updated_at = now()
    where id = c.id;

  insert into public.character_actions (
    character_id, action_type, amount, blocked_amount,
    hp_before, hp_after, armor_before, armor_after, note, performed_by
  ) values (
    c.id, p_action, p_amount, v_blocked,
    c.current_hp, v_hp, c.armor, v_armor, coalesce(p_note, ''), auth.uid()
  );
end;
$$;
revoke all on function public.apply_character_action(uuid,text,integer,text) from public;
grant execute on function public.apply_character_action(uuid,text,integer,text) to authenticated;

-- Одобрение, отклонение, архивация и восстановление персонажа.
create or replace function public.moderate_character(
  p_character_id uuid,
  p_status text,
  p_note text default ''
)
returns void
language plpgsql security definer
set search_path = ''
as $$
declare c public.characters%rowtype;
begin
  if not public.is_admin() then raise exception 'Недостаточно прав'; end if;
  if p_status not in ('approved', 'rejected', 'archived') then
    raise exception 'Недопустимый статус';
  end if;
  if char_length(coalesce(p_note, '')) > 500 then
    raise exception 'Комментарий слишком длинный';
  end if;
  select * into c from public.characters
    where id = p_character_id for update;
  if not found then raise exception 'Персонаж не найден'; end if;
  if c.status = p_status then raise exception 'Статус не изменился'; end if;

  update public.characters
    set status = p_status, updated_at = now()
    where id = p_character_id;
  insert into public.character_actions (
    character_id, action_type, amount, blocked_amount,
    hp_before, hp_after, armor_before, armor_after, note, performed_by
  ) values (
    c.id, 'status_' || p_status, 0, 0,
    c.current_hp, c.current_hp, c.armor, c.armor, coalesce(p_note, ''), auth.uid()
  );
end;
$$;
revoke all on function public.moderate_character(uuid,text,text) from public;
grant execute on function public.moderate_character(uuid,text,text) to authenticated;

-- Публичная галерея персонажей: прямые ссылки на изображения общедоступны.
insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
values ('character-images', 'character-images', true, 3145728,
        array['image/jpeg','image/png','image/webp'])
on conflict (id) do update
  set public = excluded.public,
      file_size_limit = excluded.file_size_limit,
      allowed_mime_types = excluded.allowed_mime_types;

-- Удаляем только собственные именованные политики; другие политики Storage нужно
-- проверить вручную в Supabase Dashboard (они могут расширять доступ).
drop policy if exists rpg_character_upload on storage.objects;
drop policy if exists rpg_character_read on storage.objects;
drop policy if exists rpg_character_delete on storage.objects;
create policy rpg_character_upload on storage.objects
  for insert to authenticated
  with check (
    bucket_id = 'character-images'
    and (storage.foldername(name))[1] = (select auth.uid())::text
  );
create policy rpg_character_read on storage.objects
  for select to anon, authenticated
  using (bucket_id = 'character-images');
create policy rpg_character_delete on storage.objects
  for delete to authenticated
  using (
    bucket_id = 'character-images'
    and ((storage.foldername(name))[1] = (select auth.uid())::text or public.is_admin())
  );

-- Включаем изменения в Realtime, не добавляя повторно уже включённые таблицы.
do $$
declare t text;
begin
  if exists (select 1 from pg_publication where pubname = 'supabase_realtime') then
    foreach t in array array['characters','character_cards','character_actions'] loop
      if not exists (
        select 1 from pg_publication_tables
        where pubname = 'supabase_realtime'
          and schemaname = 'public' and tablename = t
      ) then
        execute format('alter publication supabase_realtime add table public.%I', t);
      end if;
    end loop;
  end if;
end $$;
```

### `src/lib/supabase/middleware.ts`

```ts
import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";

export async function updateSession(request: NextRequest) {
  let response = NextResponse.next({ request });
  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },
        setAll(cookies) {
          cookies.forEach(({ name, value }) => request.cookies.set(name, value));
          response = NextResponse.next({ request });
          cookies.forEach(({ name, value, options }) =>
            response.cookies.set(name, value, options)
          );
        },
      },
    }
  );
  // Проверяет подпись JWT и продлевает истекающую сессию.
  await supabase.auth.getClaims();
  return response;
}
```

### `src/app/auth/confirm/route.ts`

```ts
import { NextRequest, NextResponse } from "next/server";
import { type EmailOtpType } from "@supabase/supabase-js";
import { createClient } from "@/lib/supabase/server";

export async function GET(request: NextRequest) {
  const url = new URL(request.url);
  const code = url.searchParams.get("code");
  const tokenHash = url.searchParams.get("token_hash");
  const type = url.searchParams.get("type") as EmailOtpType | null;
  const supabase = await createClient();

  if (code) {
    const { error } = await supabase.auth.exchangeCodeForSession(code);
    if (!error) return NextResponse.redirect(new URL("/dashboard", request.url));
  } else if (tokenHash && type) {
    const { error } = await supabase.auth.verifyOtp({ token_hash: tokenHash, type });
    if (!error) return NextResponse.redirect(new URL("/dashboard", request.url));
  }
  return NextResponse.redirect(new URL("/login?message=confirm-error", request.url));
}
```

```tsx
"use client";

import { useRef, useState } from "react";
import { useRouter } from "next/navigation";
import { createClient } from "@/lib/supabase/client";

const allowed: Record<string, string> = {
  "image/png": "png",
  "image/jpeg": "jpg",
  "image/webp": "webp",
};

export default function CreateCharacterForm() {
  const [name, setName] = useState("");
  const [maxHp, setMaxHp] = useState(20);
  const [image, setImage] = useState<File | null>(null);
  const [message, setMessage] = useState("");
  const [loading, setLoading] = useState(false);
  const imageInput = useRef<HTMLInputElement>(null);
  const router = useRouter();

  async function submit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    if (!name.trim() || name.trim().length > 100) { setMessage("Имя должно содержать от 1 до 100 символов"); return; }
    if (!Number.isInteger(maxHp) || maxHp < 1 || maxHp > 100000) { setMessage("HP должно быть от 1 до 100000"); return; }
    if (!image || !allowed[image.type]) { setMessage("Выберите JPEG, PNG или WebP"); return; }
    if (image.size > 3 * 1024 * 1024) { setMessage("Изображение больше 3 МБ"); return; }

    setLoading(true);
    setMessage("");
    const supabase = createClient();
    let uploadedPath: string | null = null;
    try {
      const { data: { user }, error: authError } = await supabase.auth.getUser();
      if (authError || !user) throw new Error("Нужно войти в аккаунт");

      const path = `${user.id}/${crypto.randomUUID()}.${allowed[image.type]}`;
      const { error: uploadError } = await supabase.storage
        .from("character-images")
        .upload(path, image, { contentType: image.type, upsert: false });
      if (uploadError) throw uploadError;
      uploadedPath = path;
      const { data: urlData } = supabase.storage.from("character-images").getPublicUrl(path);

      const { error } = await supabase.from("characters").insert({
        user_id: user.id,
        name: name.trim(),
        max_hp: maxHp,
        current_hp: maxHp,
        armor: 0,
        image_url: urlData.publicUrl,
        image_path: path,
        status: "pending",
      });
      if (error) throw error;
      setMessage("Заявка отправлена! Администратор увидит её в очереди модерации.");
      setName("");
      setMaxHp(20);
      setImage(null);
      if (imageInput.current) imageInput.current.value = "";
      router.refresh();
    } catch (error: unknown) {
      // Если запись в базе не создалась, не оставляем бесхозную картинку.
      if (uploadedPath) await supabase.storage.from("character-images").remove([uploadedPath]);
      setMessage(error instanceof Error ? error.message : "Ошибка отправки заявки");
    } finally {
      setLoading(false);
    }
  }

  return (
    <form onSubmit={submit} className="space-y-4">
      <label className="block text-sm font-medium">Имя персонажа
        <input className="mt-1 w-full rounded-lg border p-3" value={name} onChange={(e) => setName(e.target.value)} maxLength={100} required placeholder="Имя героя" />
      </label>
      <label className="block text-sm font-medium">Максимальное здоровье
        <input className="mt-1 w-full rounded-lg border p-3" type="number" min={1} max={100000} step={1} value={maxHp} onChange={(e) => setMaxHp(Number(e.target.value))} required />
      </label>
      <label className="block text-sm font-medium">Портрет персонажа (JPEG, PNG, WebP; до 3 МБ)
        <input ref={imageInput} className="mt-1 w-full rounded-lg border p-3" type="file" accept="image/jpeg,image/png,image/webp" required onChange={(e) => setImage(e.target.files?.[0] ?? null)} />
      </label>
      {message && <p role="status" className="rounded-lg bg-slate-100 p-3 text-sm">{message}</p>}
      <button disabled={loading} className="w-full rounded-lg bg-slate-900 p-3 font-semibold text-white disabled:opacity-50">
        {loading ? "Отправка..." : "Отправить на модерацию"}
      </button>
    </form>
  );
}
```

```tsx
import { createClient } from "@/lib/supabase/client";

export default function ModerationButtons({ id, status }: { id: string; status: CharacterStatus }) {
  const [busy, setBusy] = useState(false);
  const [message, setMessage] = useState("");
  const router = useRouter();

  async function moderate(nextStatus: "approved" | "rejected" | "archived") {
    const messageForLog = nextStatus === "rejected" ? window.prompt("Причина отклонения (необязательно):", "") : "";
    if (messageForLog === null) return;

        p_character_id: id,
        p_status: nextStatus,
        p_note: messageForLog || "",
      });
      if (error) throw error;
      router.refresh();
    } catch (error: unknown) {
      setMessage(error instanceof Error ? error.message : "Ошибка изменения статуса");
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="mt-3 flex flex-wrap items-center gap-2">
      {status !== "approved" && <button disabled={busy} onClick={() => void moderate("approved")} className="rounded-lg bg-emerald-600 px-3 py-2 text-sm font-semibold text-white disabled:opacity-50">Одобрить</button>}
      {status === "pending" && <button disabled={busy} onClick={() => void moderate("rejected")} className="rounded-lg bg-rose-100 px-3 py-2 text-sm font-semibold text-rose-800 disabled:opacity-50">Отклонить</button>}
```

```tsx
"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { createClient } from "@/lib/supabase/client";

export default function AdminCharacterActions({ character }: { character: Character }) {
  const [amount, setAmount] = useState(1);
  const [note, setNote] = useState("");
  const [busy, setBusy] = useState(false);
  const [message, setMessage] = useState("");
  const router = useRouter();

  async function apply(action: "damage" | "heal" | "set_armor") {
    if (!Number.isInteger(amount) || amount < (action === "set_armor" ? 0 : 1) || amount > 100000) {
      setMessage("Некорректное значение"); return;
    }
    setBusy(true);
    setMessage("");
    try {
      const { error } = await createClient().rpc("apply_character_action", {
        p_character_id: character.id,
        p_action: action,
        p_amount: amount,
        p_note: note.trim(),
      });
      if (error) throw error;
      setNote("");
      setMessage("Изменение сохранено в истории");
      router.refresh();
    } catch (error: unknown) {
      setMessage(error instanceof Error ? error.message : "Не удалось сохранить действие");
    } finally {
      setBusy(false);
    }
  }

  async function removeCharacter() {
    if (!window.confirm(`НАВСЕГДА удалить персонажа «${character.name}» со всеми картами и историей?`)) return;
    setBusy(true);
    setMessage("");
    const supabase = createClient();
    try {
      // Удаляем запись в БД первой; если Storage не очистится, будет только бесхозный файл,
      // а не потерянное изображение у существующего персонажа.
      const { error } = await supabase.from("characters").delete().eq("id", character.id);
      if (error) throw error;
      if (character.image_path) {
        const { error: storageError } = await supabase.storage.from("character-images").remove([character.image_path]);
        if (storageError) console.error("Изображение не удалилось из Storage", storageError);
      }
      router.replace("/admin");
      router.refresh();
    } catch (error: unknown) {
      setMessage(error instanceof Error ? error.message : "Не удалось удалить персонажа");
    } finally {
      setBusy(false);
    }
  }

  return (

      <h2 className="text-xl font-bold">Управление характеристиками</h2>
```

```tsx
    if (!name.trim() || name.trim().length > 100 || description.length > 2000) { setMessage("Проверьте длину названия и описания"); return; }
    setBusy(true); setMessage("");
    try {
      const { error } = await createClient().from("character_cards")
        .update({ name: name.trim(), description, type }).eq("id", card.id);
      if (error) throw error;
      setMessage("Сохранено"); router.refresh();
    } catch (error: unknown) { setMessage(error instanceof Error ? error.message : "Ошибка"); }
    finally { setBusy(false); }
  }

  async function remove() {
    if (!window.confirm(`Удалить карту «${card.name}»?`)) return;
    setBusy(true); setMessage("");
    try {
      const { error } = await createClient().from("character_cards").delete().eq("id", card.id);
      if (error) throw error;
      router.refresh();
    } catch (error: unknown) { setMessage(error instanceof Error ? error.message : "Ошибка"); }
    finally { setBusy(false); }
  }

  return (
```

```tsx
"use client";

import { useEffect } from "react";
import { useRouter } from "next/navigation";
import { createClient } from "@/lib/supabase/client";

export default function RealtimeRefresh() {
  const router = useRouter();

  useEffect(() => {
    const supabase = createClient();
    const channel = supabase
      .channel("rpg-live")
      .on("postgres_changes", { event: "*", schema: "public", table: "characters" }, () => router.refresh())
      .on("postgres_changes", { event: "*", schema: "public", table: "character_cards" }, () => router.refresh())
      .on("postgres_changes", { event: "*", schema: "public", table: "character_actions" }, () => router.refresh())
      .subscribe();

    // После возврата на вкладку забираем изменения, пропущенные во время сна устройства.
    const onVisible = () => {
      if (document.visibilityState === "visible") router.refresh();
    };
    document.addEventListener("visibilitychange", onVisible);
    return () => {
      document.removeEventListener("visibilitychange", onVisible);
      void supabase.removeChannel(channel);
    };
  }, [router]);
  return null;
}
```
